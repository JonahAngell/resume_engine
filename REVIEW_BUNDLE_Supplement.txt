#!/bin/sh
set -eu
root=$(git rev-parse --show-toplevel)
cd "$root"
staged=$(mktemp)
unstaged=$(mktemp)
partial=$(mktemp)
trap 'rm -f "$staged" "$unstaged" "$partial"' EXIT HUP INT TERM
git diff --cached --name-only --diff-filter=ACMR -z > "$staged"
git diff --name-only --diff-filter=ACMR -z > "$unstaged"
# Refuse to format a path that has both staged and unstaged changes.
node tools/check-partially-staged.mjs --staged "$staged" --unstaged "$unstaged" --output "$partial"
if [ -s "$partial" ]; then
  echo "Commit stopped: these files have both staged and unstaged changes:" >&2
  while IFS= read -r file; do printf '  - %s\n' "$file" >&2; done < "$partial"
  echo "Stage the entire file, stash its unstaged edits, or commit the staged portion before retrying." >&2
  exit 1
fi
if [ -s "$staged" ]; then
  xargs -0 -n 25 npx prettier --write --ignore-unknown -- < "$staged"
  xargs -0 -n 25 git add -- < "$staged"
fi
node tools/sync-app-version.mjs --hook
node tools/create-project-tree.mjs
git add -- PROJECT_TREE.txt

import os
from dataclasses import dataclass

import httpx
from fastapi import Header, HTTPException, Request, status


@dataclass(frozen=True)
class CurrentUser:
    external_id: str
    username: str
    display_name: str
    email: str | None


def _local_user() -> CurrentUser | None:
    if os.getenv("APP_ENV", "").strip().lower() != "local":
        return None
    external_id = os.getenv("DEV_USER_ID", "local-user").strip()
    username = os.getenv("DEV_USERNAME", "local.user").strip()
    if not external_id or not username:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="Local development identity is incomplete.",
        )
    return CurrentUser(
        external_id=external_id,
        username=username,
        display_name=os.getenv("DEV_DISPLAY_NAME", "Local User").strip() or username,
        email=os.getenv("DEV_EMAIL", "").strip() or None,
    )


def _bearer_fallback_enabled() -> bool:
    return os.getenv("ALLOW_BEARER_AUTH", "").strip().lower() == "true"


async def _studioshare_user(access_token: str) -> CurrentUser:
    api_url = os.getenv("STUDIOSHARE_API_URL", "").strip().rstrip("/")
    if not api_url:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="StudioShare user service is not configured.",
        )
    try:
        async with httpx.AsyncClient(timeout=10.0) as client:
            response = await client.get(
                f"{api_url}/me",
                headers={
                    "Authorization": f"Bearer {access_token}",
                    "Accept": "application/json",
                },
            )
    except httpx.HTTPError as exc:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="StudioShare user service is unavailable.",
        ) from exc
    if response.status_code in {
        status.HTTP_401_UNAUTHORIZED,
        status.HTTP_403_FORBIDDEN,
    }:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="StudioShare login is no longer valid. Sign in again.",
        )
    if not response.is_success:
        raise HTTPException(
            status_code=status.HTTP_502_BAD_GATEWAY,
            detail="StudioShare user lookup failed.",
        )
    try:
        user = response.json()
    except ValueError as exc:
        raise HTTPException(
            status_code=status.HTTP_502_BAD_GATEWAY,
            detail="StudioShare returned an invalid user response.",
        ) from exc
    if not isinstance(user, dict):
        raise HTTPException(
            status_code=status.HTTP_502_BAD_GATEWAY,
            detail="StudioShare returned an invalid user response.",
        )
    external_id = str(user.get("id") or "").strip()
    username = str(user.get("username") or "").strip()
    display_name = str(user.get("fullName") or username).strip()
    email = str(user.get("email") or "").strip() or None
    if not external_id or not username:
        raise HTTPException(
            status_code=status.HTTP_502_BAD_GATEWAY,
            detail="StudioShare user response is missing required identity fields.",
        )
    return CurrentUser(external_id, username, display_name, email)


async def resolve_current_user(
    request: Request,
    authorization: str | None = Header(default=None),
) -> CurrentUser:
    local_user = _local_user()
    if local_user is not None:
        return local_user
    access_token = request.cookies.get("access_token", "").strip()
    if not access_token and authorization and _bearer_fallback_enabled():
        scheme, separator, credential = authorization.partition(" ")
        if separator and scheme.lower() == "bearer":
            access_token = credential.strip()
    if not access_token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authenticated StudioShare identity is required.",
        )
    return await _studioshare_user(access_token)

-- Backfill only seed-imported composition rows that are visibly present in that composition payload.
INSERT INTO
  composition_items (composition_id, section, item_id, sort_order)
SELECT
  si.composition_id,
  'tool_groups',
  g.id,
  g.sort_order
FROM
  seed_imports si
  JOIN resume_compositions c ON c.id = si.composition_id
  AND c.user_id = si.user_id
  JOIN tool_groups g ON g.user_id = si.user_id
WHERE
  EXISTS (
    SELECT
      1
    FROM
      jsonb_array_elements(
        COALESCE(c.render_payload -> 'tool_groups', '[]'::jsonb)
      ) shown
    WHERE
      shown ->> 'name' = g.name
  )
ON CONFLICT (composition_id, section, item_id) DO NOTHING;

INSERT INTO
  composition_items (composition_id, section, item_id, sort_order)
SELECT
  si.composition_id,
  CASE x.kind
    WHEN 'project' THEN 'projects'
    WHEN 'employment' THEN 'work_experience'
    ELSE 'leadership'
  END,
  x.id,
  x.sort_order
FROM
  seed_imports si
  JOIN resume_compositions c ON c.id = si.composition_id
  AND c.user_id = si.user_id
  JOIN content_items x ON x.user_id = si.user_id
WHERE
  (
    x.kind = 'project'
    AND EXISTS (
      SELECT
        1
      FROM
        jsonb_array_elements(
          COALESCE(c.render_payload -> 'projects', '[]'::jsonb)
        ) shown
      WHERE
        shown ->> 'name' = x.name
        AND shown ->> 'date' = x.date_label
    )
  )
  OR (
    x.kind = 'employment'
    AND EXISTS (
      SELECT
        1
      FROM
        jsonb_array_elements(
          COALESCE(
            c.render_payload -> 'work_experience',
            '[]'::jsonb

      WHERE
        shown ->> 'organization' = x.organization
        AND shown ->> 'title' = x.title
        AND shown ->> 'date' = x.date_label

  OR (
    x.kind = 'leadership'
    AND EXISTS (
      SELECT
        1
      FROM
        jsonb_array_elements(
          COALESCE(c.render_payload -> 'leadership', '[]'::jsonb)
        ) shown
      WHERE
        shown ->> 'organization' = x.organization
        AND shown ->> 'title' = x.title
        AND shown ->> 'date' = x.date_label
    )
  )
ON CONFLICT (composition_id, section, item_id) DO NOTHING;

-- Correct the Phase 2 ownership function for composition_custom_sections,
-- which intentionally has no user_id column.
CREATE OR REPLACE FUNCTION enforce_phase2_ownership () RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  expected uuid;
BEGIN
  IF TG_TABLE_NAME = 'custom_section_entries' THEN
    SELECT user_id
    INTO expected
    FROM custom_sections
    WHERE id = NEW.section_id;

    IF expected IS NULL OR expected <> NEW.user_id THEN
      RAISE EXCEPTION 'ownership mismatch'
        USING ERRCODE = '23503';
    END IF;

  ELSIF TG_TABLE_NAME = 'composition_custom_sections' THEN
    SELECT composition.user_id
    INTO expected
    FROM resume_compositions composition
    JOIN custom_sections section
      ON section.id = NEW.section_id
     AND section.user_id = composition.user_id
    WHERE composition.id = NEW.composition_id;

    IF expected IS NULL THEN
      RAISE EXCEPTION 'ownership mismatch'
        USING ERRCODE = '23503';
    END IF;

  ELSIF TG_TABLE_NAME = 'composition_revisions' THEN
    SELECT user_id
    INTO expected
    FROM resume_compositions
    WHERE id = NEW.composition_id;

    IF expected IS NULL OR expected <> NEW.user_id THEN
      RAISE EXCEPTION 'ownership mismatch'
        USING ERRCODE = '23503';
    END IF;

  ELSE
    RAISE EXCEPTION 'unsupported ownership trigger table'
      USING ERRCODE = '23503';
  END IF;

  RETURN NEW;
END;
$$;

-- Phase 2G: additive resume metadata and revision-snapshot enrichment.
ALTER TABLE resume_compositions
ADD COLUMN display_title text NOT NULL DEFAULT '',
ADD COLUMN target_company text NOT NULL DEFAULT '',
ADD COLUMN target_role text NOT NULL DEFAULT '';

ALTER TABLE resume_compositions
ADD CONSTRAINT resume_compositions_display_title_length CHECK (length(display_title) <= 200),
ADD CONSTRAINT resume_compositions_target_company_length CHECK (length(target_company) <= 200),
ADD CONSTRAINT resume_compositions_target_role_length CHECK (length(target_role) <= 200);

UPDATE composition_revisions revision
SET
  snapshot = revision.snapshot || jsonb_build_object(
    'name',
    COALESCE(revision.snapshot ->> 'name', composition.name),
    'description',
    COALESCE(
      revision.snapshot ->> 'description',
      composition.description
    ),
    'display_title',
    COALESCE(
      revision.snapshot ->> 'display_title',
      composition.display_title
    ),
    'target_company',
    COALESCE(
      revision.snapshot ->> 'target_company',
      composition.target_company
    ),
    'target_role',
    COALESCE(
      revision.snapshot ->> 'target_role',
      composition.target_role
    )
  )
FROM
  resume_compositions composition
WHERE
  composition.id = revision.composition_id;

CREATE OR REPLACE FUNCTION enrich_composition_revision_snapshot () RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  composition resume_compositions%ROWTYPE;
BEGIN
  SELECT * INTO composition FROM resume_compositions WHERE id = NEW.composition_id;
  IF composition.id IS NULL OR composition.user_id <> NEW.user_id THEN
    RAISE EXCEPTION 'ownership mismatch' USING ERRCODE='23503';
  END IF;
  NEW.snapshot = COALESCE(NEW.snapshot, '{}'::jsonb) || jsonb_build_object(
    'name', composition.name,
    'description', composition.description,
    'display_title', composition.display_title,
    'target_company', composition.target_company,
    'target_role', composition.target_role
  );
  RETURN NEW;
END;
$$;

CREATE TRIGGER composition_revision_metadata_snapshot
BEFORE INSERT OR UPDATE ON composition_revisions FOR EACH ROW
EXECUTE FUNCTION enrich_composition_revision_snapshot ();

export type SaveState = "Unsaved" | "Saving" | "Saved" | "Error" | "Conflict";
export class SaveCoordinator<T> {
  private localVersion = 0;
  private confirmedVersion = 0;
  private chain: Promise<void> = Promise.resolve();
  private timer: number | undefined;
  constructor(
    private commit: (value: T, version: number) => Promise<void>,
    private state: (s: SaveState) => void,
  ) {}
  dirty() {
    this.localVersion += 1;
    this.state("Unsaved");
    return this.localVersion;
  }
  schedule(value: T, delay = 700) {
    const version = this.dirty();
    window.clearTimeout(this.timer);
    this.timer = window.setTimeout(() => this.enqueue(value, version), delay);
  }
  flush(value: T) {
    window.clearTimeout(this.timer);
    const version = this.localVersion || this.dirty();
    return this.enqueue(value, version);
  }
  private enqueue(value: T, version: number) {
    this.chain = this.chain.then(async () => {
      this.state("Saving");
      try {
        await this.commit(value, version);
        this.confirmedVersion = Math.max(this.confirmedVersion, version);
        this.state(
          this.confirmedVersion === this.localVersion ? "Saved" : "Unsaved",
        );
      } catch (e: any) {
        this.state(e?.status === 409 ? "Conflict" : "Error");
        throw e;
      }
    });
    return this.chain;
  }
  get unsynchronized() {
    return this.confirmedVersion !== this.localVersion;
  }
}

  return spawnSync("git", args, { cwd: root, encoding: "utf8" });
}
async function repo() {
  const root = await mkdtemp(path.join(tmpdir(), "resume-hook-"));
  for (const file of [
    "frontend/src/app-version.ts",
    "PROJECT_CONTEXT.md",
    "tools/sync-app-version.mjs",
  ]) {
    const directory = path.dirname(file);
    if (directory !== ".")
      await mkdir(path.join(root, directory), { recursive: true });
    await cp(path.join(source, file), path.join(root, file));
  }
  git(root, "init");
  git(root, "config", "user.email", "test@example.test");
  git(root, "config", "user.name", "Test");
  git(root, "add", ".");
  git(root, "commit", "-m", "baseline", "--no-verify");
  return root;
}

  const root = await repo();
  try {
    await writeFile(path.join(root, "change.txt"), "one\n");
    git(root, "add", "change.txt");
    const result = spawnSync(

});

def _render_measure_values(source: str) -> set[bool]:
    tree = ast.parse(source)
    values: set[bool] = set()

    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue

        if not isinstance(node.func, ast.Name):
            continue

        if node.func.id != "render_resume":
            continue

        if len(node.args) >= 2:
            measure_argument = node.args[1]

            if isinstance(measure_argument, ast.Constant) and isinstance(
                measure_argument.value, bool
            ):
                values.add(measure_argument.value)

        for keyword in node.keywords:
            if keyword.arg != "measure":
                continue

            if isinstance(keyword.value, ast.Constant) and isinstance(
                keyword.value.value, bool
            ):
                values.add(keyword.value.value)

    return values


def test_render_contract_remains_canonical():
    source = (ROOT / "backend/app/main.py").read_text(encoding="utf-8")

    measure_values = _render_measure_values(source)

    assert True in measure_values, (
        "Browser preview must call render_resume with measurement enabled."
    )
    assert False in measure_values, (
        "PDF export must call render_resume with measurement disabled."
    )

    assert source.count("render_resume(") >= 2

import json
import os
from pathlib import Path

import psycopg
import pytest

async def apply_migrations(conn, names=None):
    paths = sorted((ROOT / "backend/migrations").glob("*.sql"))
    if names is not None:
        paths = [path for path in paths if path.name in names]
    for path in paths:
        await conn.execute(path.read_text(encoding="utf-8"))


async def exercise_phase2():
    schema = f"phase2_{uuid4().hex}"
    async with await psycopg.AsyncConnection.connect(URL, autocommit=True) as admin:
        await admin.execute(f'CREATE SCHEMA "{schema}"')
    try:
        options = f"-c search_path={schema},public"
        async with await psycopg.AsyncConnection.connect(URL, options=options) as conn:
            await apply_migrations(
                conn,
                {
                    "001_phase1b_master_library.sql",
                    "002_phase1c_editing.sql",
                    "003_phase1d_ownership_guards.sql",
                },
            )
            user_a = await ensure_user(
                conn, CurrentUser("phase2-a", "a", "User A", None)
            )
            user_b = await ensure_user(
                conn, CurrentUser("phase2-b", "b", "User B", None)
            )
            imported, composition_id = await import_seed(
                conn,
                user_a["id"],
                ROOT / "seed/jonah-resume.json",
                "phase2-seed",
            )
            assert imported

            # Read only columns available in the Phase 1D schema. The Phase 2
            # repository loader cannot be used until migration 004 adds description.
            before = await (
                await conn.execute(
                    "SELECT id, name, revision, render_payload, archived_at "
                    "FROM resume_compositions "
                    "WHERE id=%s AND user_id=%s AND archived_at IS NULL",
                    (composition_id, user_a["id"]),
                )
            ).fetchone()

            assert before is not None
            original_payload = before[3]

            # Apply Phase 2 to populated Phase 1D data and verify backfill compatibility.
            await apply_migrations(conn, {"004_phase2_historical_library.sql"})
            after = await load_composition(conn, user_a["id"], composition_id)
            assert after["render_payload"] == original_payload
            migrated = await list_revisions(conn, user_a["id"], composition_id)
            assert [row["revision"] for row in migrated] == [1]
            migration_revision = await get_revision(
                conn, user_a["id"], composition_id, migrated[0]["id"]
            )
            assert migration_revision["snapshot"]["resume"] == original_payload

            # A successful save creates immutable history; a stale save changes nothing.
            changed = json.loads(json.dumps(original_payload))
            changed["summary"] = "Phase 2 integration summary"
            saved = await save_composition(
                conn, user_a["id"], composition_id, 1, changed, "integration"
            )
            assert saved["revision"] == 2
            assert (
                await save_composition(
                    conn, user_a["id"], composition_id, 1, original_payload, "stale"
                )
                is None
            )
            revisions = await list_revisions(conn, user_a["id"], composition_id)
            assert [row["revision"] for row in revisions] == [2, 1]

            # Restore creates revision 3 and preserves revisions 1 and 2.
            restored, error = await restore_revision(
                conn,
                user_a["id"],
                composition_id,
                migrated[0]["id"],
                2,
            )
            assert error is None
            assert restored["revision"] == 3
            assert restored["render_payload"] == original_payload
            assert [
                row["revision"]
                for row in await list_revisions(conn, user_a["id"], composition_id)
            ] == [3, 2, 1]

            # Stale and foreign restoration attempts must not mutate current state.
            stale, error = await restore_revision(
                conn,
                user_a["id"],
                composition_id,
                migrated[0]["id"],
                2,
            )
            assert stale is None and error == "stale"
            foreign, error = await restore_revision(
                conn,
                user_b["id"],
                composition_id,
                migrated[0]["id"],
                3,
            )
            assert foreign is None and error == "missing"
            current = await load_composition(conn, user_a["id"], composition_id)
            assert current["revision"] == 3

            # Search and archive operations remain scoped to the current user.
            own_results = await search(conn, user_a["id"], "Seed", "all", 50, 0)
            foreign_results = await search(conn, user_b["id"], "Seed", "all", 50, 0)
            assert any(row["type"] == "composition" for row in own_results)
            assert foreign_results == []
            summary_id = (await fetch_library(conn, user_a["id"]))["summary_variants"][
                0
            ]["id"]
            assert (
                await set_archive(
                    conn, user_b["id"], "summary_variants", summary_id, True
                )
                is None
            )
            assert await set_archive(
                conn, user_a["id"], "summary_variants", summary_id, True
            )
            archived = await search(conn, user_a["id"], "Seed", "archived", 50, 0)
            assert any(row["id"] == summary_id and row["archived"] for row in archived)
            assert await set_archive(
                conn, user_a["id"], "summary_variants", summary_id, False
            )

            # Database guards reject cross-user custom relationships and revisions.
            section_id = (
                await (
                    await conn.execute(
                        "INSERT INTO custom_sections(user_id,title,sort_order) "
                        "VALUES(%s,'Training',0) RETURNING id",
                        (user_a["id"],),
                    )
                ).fetchone()
            )[0]
            with pytest.raises(psycopg.Error):
                async with conn.transaction():
                    await conn.execute(
                        "INSERT INTO custom_section_entries"
                        "(user_id,section_id,primary_text,sort_order) VALUES(%s,%s,'x',0)",
                        (user_b["id"], section_id),
                    )
            with pytest.raises(psycopg.Error):
                async with conn.transaction():
                    await conn.execute(
                        "INSERT INTO composition_revisions"
                        "(composition_id,user_id,revision,snapshot) VALUES(%s,%s,99,'{}'::jsonb)",
                        (composition_id, user_b["id"]),
                    )
    finally:
        async with await psycopg.AsyncConnection.connect(URL, autocommit=True) as admin:
            await admin.execute(f'DROP SCHEMA IF EXISTS "{schema}" CASCADE')

#!/usr/bin/env node
import { readFile, writeFile } from "node:fs/promises";
import process from "node:process";

function parseArgs(args) {
  const options = {};
  for (let index = 0; index < args.length; index += 2) {
    const key = args[index];
    const value = args[index + 1];
    if (!["--staged", "--unstaged", "--output"].includes(key) || !value) {
      throw new Error(
        "Usage: check-partially-staged.mjs --staged <nul-file> --unstaged <nul-file> --output <text-file>",
      );
    }
    options[key.slice(2)] = value;
  }
  if (!options.staged || !options.unstaged || !options.output) {
    throw new Error("Missing required staged, unstaged, or output path.");
  }
  return options;
}

function parseNul(buffer) {
  return buffer.toString("utf8").split("\0").filter(Boolean);
}

async function main() {
  const options = parseArgs(process.argv.slice(2));
  const staged = new Set(parseNul(await readFile(options.staged)));
  const unstaged = new Set(parseNul(await readFile(options.unstaged)));
  const partial = [...staged]
    .filter((file) => unstaged.has(file))
    .sort((a, b) => a.localeCompare(b));
  await writeFile(
    options.output,
    partial.length ? `${partial.join("\n")}\n` : "",
    "utf8",
  );
}

main().catch((error) => {
  console.error(`Partial-stage check failed: ${error.message}`);
  process.exitCode = 1;
});

#!/usr/bin/env node
import { readdir, writeFile } from "node:fs/promises";
import path from "node:path";

const OUTPUT = "PROJECT_TREE.txt";
const EXCLUDED_NAMES = new Set([
  ".git",
  ".venv",
  "node_modules",
  "dist",
  "__pycache__",
  ".pytest_cache",
  "playwright-report",
  "test-results",
  "htmlcov",
  ".coverage",
  ".env",
  "REVIEW_BUNDLE.txt",
  OUTPUT,
]);

async function entries(directory) {
  return (await readdir(directory, { withFileTypes: true }))
    .filter(
      (entry) =>
        !EXCLUDED_NAMES.has(entry.name) && !entry.name.endsWith(".zip"),
    )
    .sort((left, right) => {
      if (left.isDirectory() !== right.isDirectory())
        return left.isDirectory() ? -1 : 1;
      return left.name.localeCompare(right.name);
    });
}

async function build(directory, prefix = "") {
  const children = await entries(directory);
  const lines = [];
  for (let index = 0; index < children.length; index += 1) {
    const child = children[index];
    const last = index === children.length - 1;
    lines.push(`${prefix}${last ? "└── " : "├── "}${child.name}`);
    if (child.isDirectory()) {
      lines.push(
        ...(await build(
          path.join(directory, child.name),
          `${prefix}${last ? "    " : "│   "}`,
        )),
      );
    }
  }
  return lines;
}

await writeFile(OUTPUT, `${[".", ...(await build("."))].join("\n")}\n`, "utf8");
console.log(`Updated ${OUTPUT}`);

  };
  for (let i = 0; i < args.length; i += 1) {
    const arg = args[i];
    if (arg === "--files") result.manifest = args[++i];
    else if (arg === "--output") result.output = args[++i];
    else if (arg === "--title") result.title = args[++i];
    else if (arg === "--help" || arg === "-h") {
      console.log(
        "Usage: node tools/create-review-bundle.mjs [--files manifest] [--output file] [--title text] [file ...]",
      );
      process.exit(0);
    } else if (arg.startsWith("--")) throw new Error(`Unknown option: ${arg}`);
    else result.files.push(arg);
    if (
      (arg === "--files" || arg === "--output" || arg === "--title") &&
      !args[i]
    )
      throw new Error(`${arg} requires a value.`);
  }
  return result;
}

function git(args) {
  try {
    return execFileSync("git", args, {
      encoding: "utf8",
      stdio: ["ignore", "pipe", "ignore"],
    }).trimEnd();
  } catch {
    return null;
  }
}

function isBinary(buffer) {
  if (buffer.includes(0)) return true;
  const sample = buffer.subarray(0, Math.min(buffer.length, 8192));
  let suspicious = 0;
  for (const byte of sample)
    if (byte < 7 || (byte > 13 && byte < 32)) suspicious += 1;
  return sample.length > 0 && suspicious / sample.length > 0.03;
}

function relativePosix(root, absolute) {
  return path.relative(root, absolute).split(path.sep).join("/");
}

function assertSafeRelative(relative) {
  if (
    !relative ||
    relative === "." ||
    relative.startsWith("../") ||
    path.isAbsolute(relative)
  )
    throw new Error(`Path escapes repository: ${relative || "[empty]"}`);
  const parts = relative.split("/");
  if (
    parts.some((part) => BLOCKED_PARTS.has(part.toLowerCase())) ||
    BLOCKED_NAMES.has(parts.at(-1).toLowerCase()) ||
    parts.at(-1).toLowerCase().startsWith(".env.")
  ) {
    throw new Error(`Sensitive or generated path is not allowed: ${relative}`);
  }
}

async function manifestFiles(file, root) {
  const absolute = path.resolve(root, file);
  const relative = relativePosix(root, absolute);
  assertSafeRelative(relative);
  return (await readFile(absolute, "utf8"))
    .split(/\r?\n/u)
    .map((line) => line.trim())
    .filter((line) => line && !line.startsWith("#"));
}

async function collect(requested, root, outputAbsolute) {
  const results = [];
  let total = 0;
  for (const supplied of [...new Set(requested)]) {
    const absolute = path.resolve(root, supplied);
    const relative = relativePosix(root, absolute);
    assertSafeRelative(relative);
    if (absolute === outputAbsolute)
      throw new Error("The output file cannot also be an input.");
    let canonical;
    try {
      canonical = await realpath(absolute);
    } catch {
      throw new Error(`Requested file does not exist: ${relative}`);
    }
    const canonicalRelative = relativePosix(root, canonical);
    assertSafeRelative(canonicalRelative);
    if (
      canonicalRelative.startsWith("../") ||
      path.isAbsolute(canonicalRelative)
    )
      throw new Error(`Resolved path escapes repository: ${relative}`);
    const info = await stat(canonical);
    if (!info.isFile())
      throw new Error(`Requested path is not a regular file: ${relative}`);
    if (info.size > MAX_FILE_BYTES)
      throw new Error(`File exceeds ${MAX_FILE_BYTES} byte limit: ${relative}`);
    const buffer = await readFile(canonical);
    if (isBinary(buffer))
      throw new Error(`Binary content is not allowed: ${relative}`);
    total += buffer.length;
    if (total > MAX_BUNDLE_BYTES)
      throw new Error(`Bundle inputs exceed ${MAX_BUNDLE_BYTES} byte limit.`);
    results.push({
      relative,
      buffer,
      sha: createHash("sha256").update(buffer).digest("hex"),
    });
  }
  return results;
}

function divider() {
  return "=".repeat(80);
}
function build(title, files, root) {
  const d = divider();
  const commit = git(["rev-parse", "HEAD"]) || "[Unavailable]";
  const branch =
    git(["branch", "--show-current"]) || "[Detached HEAD or unavailable]";
  const status = git(["status", "--short"]) || "[Clean or unavailable]";
  const lines = [
    title,
    `Generated: ${new Date().toISOString()}`,
    `Project root: ${root}`,
    "",
    d,
    "REPOSITORY FINGERPRINT",
    d,
    `Commit: ${commit}`,
    `Branch: ${branch}`,
    "Working tree status:",
    status,
    "",
    d,
    "FILE INVENTORY AND SHA-256",
    d,
  ];
  for (const file of files) lines.push(`${file.sha}  ${file.relative}`);
  for (const file of files) {
    lines.push(
      "",
      d,
      `BEGIN FILE: ${file.relative}`,
      d,
      "",
      file.buffer.toString("utf8"),
    );
    if (file.buffer.length && !file.buffer.toString("utf8").endsWith("\n"))
      lines.push("");
    lines.push(d, `END FILE: ${file.relative}`, d);
  }
  return `${lines.join("\n")}\n`;
}

async function main() {
  const options = parseArgs(process.argv.slice(2));
  const root = await realpath(process.cwd());
  const output = path.resolve(root, options.output);
  const outputRelative = relativePosix(root, output);
  assertSafeRelative(outputRelative);
  let requested = [...options.files];
  if (options.manifest)
    requested = [
      ...(await manifestFiles(options.manifest, root)),
      ...requested,
    ];
  if (!requested.length) throw new Error("No files requested.");
  const files = await collect(requested, root, output);
  const bundle = build(options.title, files, root);
  if (Buffer.byteLength(bundle) > MAX_BUNDLE_BYTES)
    throw new Error(`Generated bundle exceeds ${MAX_BUNDLE_BYTES} byte limit.`);
  await writeFile(output, bundle, "utf8");
  console.log(`Created ${outputRelative} with ${files.length} file(s).`);
}

main().catch((error) => {
  console.error(`Review bundle failed: ${error.message}`);
  process.exitCode = 1;
});

import { readFile, rename, unlink, writeFile } from "node:fs/promises";
import { spawnSync } from "node:child_process";
import path from "node:path";

const VERSION_SOURCE = "frontend/src/app-version.ts";
const VERSION_FILES = [VERSION_SOURCE, "PROJECT_CONTEXT.md"];
const VERSION_PATTERN = /^v(\d+)\.(\d+)\.(\d+)$/;
const SOURCE_PATTERN = /^export const RESUME_ENGINE_VERSION = "([^"]+)";$/gm;
const TARGET = {
  file: "PROJECT_CONTEXT.md",
  id: "PROJECT-CONTEXT",
  render: (version) => `**Current version:** ${version}`,
};

function fail(message) {
  throw new Error(message);
}
function escapeRegExp(value) {
  return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}
function readAuthoritativeVersion(source) {
  const matches = [...source.matchAll(SOURCE_PATTERN)];
  if (matches.length !== 1)
    fail(
      `${VERSION_SOURCE}: expected exactly one RESUME_ENGINE_VERSION declaration; found ${matches.length}.`,
    );
  if (!VERSION_PATTERN.test(matches[0][1]))
    fail(
      `${VERSION_SOURCE}: invalid RESUME_ENGINE_VERSION "${matches[0][1]}".`,
    );
  return matches[0][1];
}
function nextVersion(version) {
  const match = version.match(VERSION_PATTERN);
  if (!match)
    fail(
      `${VERSION_SOURCE}: ${version} cannot be bumped; expected vMAJOR.MINOR.PATCH.`,
    );
  return `v${match[1]}.${match[2]}.${Number(match[3]) + 1}`;
}
function replaceSourceVersion(source, version) {
  const matches = [...source.matchAll(SOURCE_PATTERN)];
  if (matches.length !== 1)
    fail(
      `${VERSION_SOURCE}: expected exactly one declaration; found ${matches.length}.`,
    );
  return source.replace(
    SOURCE_PATTERN,
    `export const RESUME_ENGINE_VERSION = "${version}";`,
  );
}
function replaceManagedRegion(content, version) {
  const begin = `<!-- RESUME-ENGINE-VERSION:${TARGET.id}:BEGIN -->`;
  const end = `<!-- RESUME-ENGINE-VERSION:${TARGET.id}:END -->`;
  const pattern = new RegExp(
    `${escapeRegExp(begin)}\\r?\\n([\\s\\S]*?)\\r?\\n${escapeRegExp(end)}`,
    "g",
  );
  const matches = [...content.matchAll(pattern)];
  if (matches.length !== 1)
    fail(
      `${TARGET.file}: expected exactly one managed region ${TARGET.id}; found ${matches.length}.`,
    );
  const expected = TARGET.render(version);
  return {
    content: content.replace(pattern, `${begin}\n${expected}\n${end}`),
    current: matches[0][1],
    expected,
  };
}
async function atomicWrite(file, content) {
  const temporary = `${file}.resume-version-${process.pid}.tmp`;
  try {
    await writeFile(temporary, content, "utf8");
    await rename(temporary, file);
  } catch (error) {
    await unlink(temporary).catch(() => {});
    throw error;
  }
}
function git(root, args, { allowFailure = false } = {}) {
  const result = spawnSync("git", args, { cwd: root, encoding: "utf8" });
  if (result.status !== 0 && !allowFailure)
    fail(
      `git ${args.join(" ")} failed: ${(result.stderr || result.stdout).trim()}`,
    );
  return result;
}
function gitPath(root, name) {
  return path.resolve(
    root,
    git(root, ["rev-parse", "--git-path", name]).stdout.trim(),
  );
}
async function exists(file) {
  try {
    await readFile(file);
    return true;
  } catch {
    return false;
  }
}
async function inProgressOperation(root) {
  for (const name of [
    "MERGE_HEAD",
    "CHERRY_PICK_HEAD",
    "REVERT_HEAD",
    "rebase-merge",
    "rebase-apply",
  ])
    if (await exists(gitPath(root, name))) return name;
  return null;
}
async function readVersionFromHead(root) {
  const result = git(root, ["show", `HEAD:${VERSION_SOURCE}`], {
    allowFailure: true,
  });
  return result.status === 0 ? readAuthoritativeVersion(result.stdout) : null;
}
async function loadPlan(root, version, { replaceSource = false } = {}) {
  const originals = new Map();
  for (const file of VERSION_FILES)
    originals.set(file, await readFile(path.join(root, file), "utf8"));
  const outputs = new Map(originals);
  if (replaceSource)
    outputs.set(
      VERSION_SOURCE,
      replaceSourceVersion(outputs.get(VERSION_SOURCE), version),
    );
  const region = replaceManagedRegion(outputs.get(TARGET.file), version);
  outputs.set(TARGET.file, region.content);
  return {
    originals,
    outputs,
    drift:
      region.current === region.expected
        ? []
        : [`${TARGET.file} (${TARGET.id})`],
  };
}
async function writePlan(root, plan) {
  for (const [file, content] of plan.outputs)
    if (content !== plan.originals.get(file))
      await atomicWrite(path.join(root, file), content);
}
function stagedNonVersionFiles(root) {
  const result = git(root, [
    "diff",
    "--cached",
    "--name-only",
    "--diff-filter=ACMR",
  ]);
  const managed = new Set(VERSION_FILES);
  return result.stdout
    .split(/\r?\n/)
    .filter(Boolean)
    .filter((file) => !managed.has(file) && file !== "PROJECT_TREE.txt");
}
async function hookMode(root) {
  const operation = await inProgressOperation(root);
  if (operation) {
    console.log(
      `Skipping automatic application-version bump during ${operation}.`,
    );
    return;
  }
  if (/amend/i.test(process.env.GIT_REFLOG_ACTION || "")) {
    console.log("Skipping automatic application-version bump during amend.");
    return;
  }
  if (stagedNonVersionFiles(root).length === 0) {
    console.log(
      "Skipping automatic application-version bump because no eligible non-version change is staged.",
    );
    return;
  }
  const source = await readFile(path.join(root, VERSION_SOURCE), "utf8");
  const current = readAuthoritativeVersion(source);
  const head = await readVersionFromHead(root);
  if (!head) fail("Cannot determine the application version from HEAD.");
  const expected = nextVersion(head);
  if (current !== head && current !== expected)
    fail(
      `Working version ${current} is not HEAD ${head} or its next version ${expected}; review manually.`,
    );
  const target = current === head ? expected : current;
  const plan = await loadPlan(root, target, {
    replaceSource: current === head,
  });
  await writePlan(root, plan);
  const check = await loadPlan(root, target);
  if (check.drift.length)
    fail(`Version remains out of sync: ${check.drift.join(", ")}.`);
  git(root, ["add", "--", ...VERSION_FILES]);
  console.log(
    current === head
      ? `Prepared and staged application version ${target}.`
      : `Reused and staged prepared application version ${target}.`,
  );
}
async function main() {
  const args = process.argv.slice(2);
  const root = process.cwd();
  const modes = [
    args.includes("--check"),
    args.includes("--bump"),
    args.includes("--hook"),
  ].filter(Boolean).length;
  if (
    modes > 1 ||
    args.some((arg) => !["--check", "--bump", "--hook"].includes(arg))
  )
    fail("Usage: node tools/sync-app-version.mjs [--check|--bump|--hook]");
  if (args.includes("--hook")) return hookMode(root);
  const source = await readFile(path.join(root, VERSION_SOURCE), "utf8");
  const current = readAuthoritativeVersion(source);
  const version = args.includes("--bump") ? nextVersion(current) : current;
  const plan = await loadPlan(root, version, {
    replaceSource: args.includes("--bump"),
  });
  if (args.includes("--check")) {
    if (plan.drift.length)
      fail(
        `Application version ${version} is out of sync in: ${plan.drift.join(", ")}. Run npm run version:sync.`,
      );
    console.log(`Application version ${version} is synchronized.`);
    return;
  }
  await writePlan(root, plan);
  console.log(
    args.includes("--bump")
      ? `Bumped and synchronized application version ${current} -> ${version}.`
      : `Application version ${version} synchronized.`,
  );
}
main().catch((error) => {
  console.error(`Version synchronization failed: ${error.message}`);
  process.exitCode = 1;
});
